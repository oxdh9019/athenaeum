```python
class WorldCharacterCoordinator:
    """世界与角色协调器"""
    
    def coordinate(
        self,
        world: WorldConfig,
        characters: List[Character]
    ) -> CoordinatedWorld:
        """协调世界设定与角色设定"""
        
        coordinated = {
            "world": world,
            "characters": [],
            "relationships": [],
            "timeline": []
        }
        
        # 1. 职业-地点映射
        occupation_to_location = self._map_occupations(world, characters)
        
        # 2. 年龄-世界时间对齐
        for char in characters:
            char.birth_year = world.start_year - char.age
            char.occupation_history = self._generate_occupation_history(
                char, world
            )
        
        # 3. 社会关系生成
        coordinated["relationships"] = self._generate_social_network(
            world, characters
        )
        
        # 4. 共同历史事件
        coordinated["timeline"] = self._generate_shared_events(
            world, characters
        )
        
        return CoordinatedWorld(**coordinated)
    
    def _generate_social_network(
        self,
        world: WorldConfig,
        characters: List[Character]
    ) -> List[Relationship]:
        """生成社会关系网络"""
        relationships = []
        
        for char_a, char_b in combinations(characters, 2):
            # 分析潜在联系
            potential = self._analyze_potential_connection(
                world, char_a, char_b
            )
            
            if potential.strength > 0.3:
                # 生成历史交集
                history = self._generate_shared_history(
                    char_a, char_b, potential
                )
                
                relationships.append(Relationship(
                    from_char=char_a.id,
                    to_char=char_b.id,
                    type=potential.type,
                    strength=potential.strength,
                    shared_history=history
                ))
        
        return relationships
```

```python
        "publish_book": {
            "reputation_change": 0.1,  # 角色声誉

        },
        
        "conflict": {
            "relationship_change": -0.3,  # 关系变化
            "location_tension": 0.2,      # 地点紧张度
            "gossip_spread": True         # 八卦传播
        },
        
        "collaboration": {
            "relationship_change": 0.2,
            "joint_opportunities": True,
            "social_network_boost": 0.1
        },
        
        "move_location": {
            "population_shift": True,
            "location_activity": 0.1,
            "old_location_activity": -0.05
        }
    }
    
    def propagate_influence(
        self,
        character: Character,
        event: CharacterEvent
    ) -> List[WorldStateChange]:
        """传播角色事件到世界"""
        
        changes = []
        influence_type = self.INFLUENCE_TYPES.get(event.type, {})
        
        # 1. 直接影响
        changes.append(WorldStateChange(
            type="reputation",
            target=character.id,
            delta=influence_type.get("reputation_change", 0)
        ))
        
        # 2. 涟漪效应 - 关联角色
        related_chars = self._get_related_characters(character)
        for related in related_chars:
            changes.append(self._generate_ripple_effect(
                related, character, event
            ))
        
        # 3. 地点影响
        changes.append(WorldStateChange(
            type="location_fame",
            target=character.location,
            delta=influence_type.get("location_fame", 0)
        ))
        
        # 4. 八卦传播
        if influence_type.get("gossip_spread"):
            changes.extend(self._spread_gossip(character, event))
        
        return changes
```

```python
class WeatherSystem:
    """天气系统"""
    
    WEATHER_TYPES = {
        "clear": {
            "threat_response_delta": 0,
            "comfort_state_delta": 0.1,
            "movement_modifier": 1.0,
            "activity_boost": ["outdoor", "market"]
        },
        "rain": {
            "threat_response_delta": 0.1,
            "comfort_state_delta": -0.1,
            "movement_modifier": 0.8,
            "activity_boost": ["indoor", "tavern"]
        },
        "storm": {
            "threat_response_delta": 0.25,
            "comfort_state_delta": -0.2,
            "movement_modifier": 0.5,
            "activity_boost": ["shelter"],
            "forced_indoor": True
        },
        "festival_clear": {
            "threat_response_delta": -0.1,
            "comfort_state_delta": 0.2,
            "social_approval_delta": 0.1,
            "movement_modifier": 1.2,
            "activity_boost": ["social", "celebration"]
        }
    }
    
    def tick(self, world_state: WorldState) -> WeatherState:
        """更新天气状态"""
        
        # 1. 基于季节计算基础概率
        season = world_state.current_season
        base_prob = self._get_season_base_probability(season)
        
        # 2. 考虑前一天的天气
        yesterday = world_state.weather_history[-1] if world_state.weather_history else "clear"
        transition_prob = self._get_transition_probability(yesterday)
        
        # 3. 生成新天气
        new_weather = self._roll_weather(base_prob, transition_prob)
        
        # 4. 计算对角色的影响
        weather_effects = self.WEATHER_TYPES[new_weather]
        
        return WeatherState(
            type=new_weather,
            effects=weather_effects,
            duration=self._roll_duration(new_weather)
        )
    
    def get_character_effects(
        self,
        character: Character,
        weather: WeatherState
    ) -> Dict[str, float]:
        """计算天气对特定角色的影响"""
        
        effects = {}
        weather_effects = weather.effects
        
        # 基础影响
        effects["threat_response"] = weather_effects.get("threat_response_delta", 0)
        effects["comfort_state"] = weather_effects.get("comfort_state_delta", 0)
        
        # 角色特质调节
        if character.personality.extraversion > 0.7:
            if "rain" in weather.type:
                effects["comfort_state"] -= 0.1
            if weather.type == "festival_clear":
                effects["comfort_state"] += 0.15
        
        if character.personality.neuroticism > 0.6:
            if weather.type == "storm":
                effects["threat_response"] += 0.15
        
        return effects
```

### 4.2 World Agent 架构

```python
class WorldAgent:
    """
    世界 Agent - 管理世界作为一个"有机整体"的运行
    """
    
    def __init__(self, config: WorldConfig):
        self.config = config
        
        # 子系统
        self.nature = NatureSystem()      # 自然规律
        self.society = SocietySystem()    # 社会规律
        self.events = EventGenerator()     # 事件生成
        self.economy = EconomicSystem()   # 经济系统
        self.politics = PoliticalSystem()  # 政治系统
        
        # 世界状态
        self.state = WorldState()
        
        # "世界意志"配置
        self.world_will = config.world_will
    
    async def think(self, tick: int) -> WorldDecision:
        """
        世界"思考" - 每个 Tick 运行
        决定世界层面发生什么
        """
        
        decisions = []
        
        # 1. 自然系统思考
        nature_decision = await self.nature.think(self.state)
        decisions.append(nature_decision)
        
        # 2. 社会系统思考
        society_decision = await self.society.think(
            self.state,
            self._get_collective_mood()
        )
        decisions.append(society_decision)
        
        # 3. 事件生成
        event_decision = await self.events.think(
            self.state,
            self.world_will.narrative_bias
        )
        decisions.append(event_decision)
        
        # 4. 整合决策
        final_decision = self._synthesize_decisions(decisions)
        
        return final_decision
    
    async def _get_collective_mood(self) -> CollectiveMood:
        """计算集体情绪 - 基于所有角色的平均状态"""
        
        characters = await self.state.get_all_characters()
        
        avg_threat = mean([c.desire_state.threat_response for c in characters])
        avg_comfort = mean([c.desire_state.comfort_state for c in characters])
        avg_social = mean([c.desire_state.social_approval for c in characters])
        
        return CollectiveMood(
            tension=avg_threat,
            happiness=avg_comfort,
            cohesion=avg_social
        )
```

### 4.3 社会规律 Agent

```python
class SocietySystem:
    """社会规律 - 世界"社会意志"的体现"""
    
    def __init__(self):
        self.rules = SocialRules()
        self.trends = []  # 当前社会趋势
    
    async def think(
        self,
        world_state: WorldState,
        collective_mood: CollectiveMood
    ) -> SocietyDecision:
        """
        社会系统思考 - 决定社会层面的发展
        """
        
        # 1. 评估当前社会健康度
        health = self._evaluate_society_health(
            world_state, collective_mood
        )
        
        # 2. 生成/推进趋势
        new_trends = self._generate_trends(
            world_state, health
        )
        
        # 3. 决定是否触发社会事件
        should_trigger = self._should_trigger_social_event(
            health, new_trends
        )
        
        if should_trigger:
            return self._create_social_event(
                world_state, health, new_trends
            )
        
        return SocietyDecision(
            type="maintain",
            trend_updates=new_trends
        )
    
    def _generate_trends(self, state, health) -> List[Trend]:
        """生成社会趋势"""
        trends = []
        
        # 经济增长趋势
        if state.economic.activity > 0.7:
            trends.append(Trend(
                name="prosperity",
                strength=0.8,
                direction="up"
            ))
        
        # 社会紧张趋势
        if collective_mood.tension > 0.6:
            trends.append(Trend(
                name="social_tension",
                strength=0.6,
                direction="up"
            ))
        
        # 文化繁荣趋势
        if state.cultural.activity > 0.5:
            trends.append(Trend(
                name="cultural_renaissance",
                strength=0.5,
                direction="stable"
            ))
        
        return trends
```

## 五、总结

### 设计决策总结

| 问题 | 设计决策 |
|------|----------|
| **初始化协调** | WorldX 先生成骨架，再协调角色生成，最后编织关系 |
| **角色影响世界** | 涟漪效应系统，自动传播到关联角色和地点 |
| **世界自主发展** | 独立事件系统（天气、季节、经济、社会） |
| **World Agent** | 元 Agent 设计：社会规律 + 自然系统 + 事件生成 |

### 核心设计理念

1. **世界不是被动容器，而是活的系统**
2. **角色与世界相互影响，形成反馈循环**
3. **世界"意志"可配置，允许不同风格的世界**
